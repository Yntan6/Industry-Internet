import json
from common.Logger import logger
from quickfaas.remotebus import publish

def main(message, cloudName):
    logger.debug("publish topic:%s, payload: %s, cloudName: %s" % (__topic__, message, cloudName))

    # 转换 original message 到 target format
    transformed_data = { measure['name']: measure['value'] for measure in message['measures'] }
    
    # 发布转换后的数据
    payload = json.dumps(transformed_data)
    logger.debug("transformed payload: %s" % payload)
    publish(__topic__, payload, __qos__, cloud_name=cloudName)
